import {spawn} from 'node:child_process';
import {readFileSync,statSync} from 'node:fs';
import {homedir} from 'node:os';
import path from 'node:path';
import {extractNotes} from './public/music.mjs';
export class Composer {
 constructor(cwd){this.cwd=cwd;this.pending=new Map();this.sequence=0;this.job=null;this.ready=null;}
 async connect(){
  if(this.ready)return this.ready;
  this.ready=this.initialize().catch(e=>{this.ready=null;throw e;});return this.ready;
 }
 async initialize(){
  const env=Object.fromEntries(['PATH','HOME','CODEX_HOME','USER','TMPDIR','LANG'].filter(k=>process.env[k]).map(k=>[k,process.env[k]]));
  const args=['app-server'];
  for(const key of ['shell_tool','unified_exec','apps','plugins','multi_agent','hooks','browser_use','computer_use','image_generation','view_image'])args.push('-c',`features.${key}=false`);
  args.push('-c','features.skip_host_skill_discovery=true');
  try{const s=readFileSync(path.join(process.env.CODEX_HOME||path.join(homedir(),'.codex'),'config.toml'),'utf8');for(const m of s.matchAll(/^\[mcp_servers\.(?:"([^"]+)"|([A-Za-z0-9_-]+))\]/gm))args.push('-c',`mcp_servers.${(m[1]||m[2])}.enabled=false`);}catch{}
  this.proc=spawn(process.env.CODEX_PATH||'codex',args,{cwd:this.cwd,env,stdio:['pipe','pipe','pipe']});
  const processInstance=this.proc;
  this.proc.stderr.resume();let buffer='';
  this.proc.stdout.on('data',data=>{buffer+=data;let i;while((i=buffer.indexOf('\n'))>=0){const line=buffer.slice(0,i);buffer=buffer.slice(i+1);try{this.message(JSON.parse(line));}catch{}}});
  const fail=()=>{if(this.proc!==processInstance)return;for(const p of this.pending.values()){clearTimeout(p.timer);p.reject(Error('Codex disconnected'));}this.pending.clear();this.job?.finish(Error('Codex disconnected'));this.ready=null;};
  this.proc.on('error',fail);this.proc.on('exit',fail);
  await this.rpc('initialize',{clientInfo:{name:'find_the_beat',version:'0.1.0'},capabilities:{experimentalApi:true}});
  this.send({method:'initialized',params:{}});
 }
 send(msg){this.proc.stdin.write(JSON.stringify(msg)+'\n');}
 rpc(method,params){const id=++this.sequence;return new Promise((resolve,reject)=>{const timer=setTimeout(()=>{this.pending.delete(id);reject(Error('Codex connection timed out'));},15000);this.pending.set(id,{resolve,reject,timer});this.send({id,method,params});});}
 message(m){
  if(m.id!==undefined&&this.pending.has(m.id)){const p=this.pending.get(m.id);this.pending.delete(m.id);clearTimeout(p.timer);m.error?p.reject(Error(m.error.message)):p.resolve(m.result);return;}
  if(m.id!==undefined&&m.method){this.send({id:m.id,error:{code:-32601,message:'Tools and approvals are disabled for the music composer'}});return;}
  const j=this.job;if(!j||m.params?.threadId!==j.threadId)return;
  if(m.method==='item/agentMessage/delta'){j.text+=m.params.delta;const notes=extractNotes(j.text);while(j.sent<notes.length)j.emit({type:'note',event:notes[j.sent++],elapsedMs:Math.round(performance.now()-j.start)});}
  if(m.method==='turn/completed')j.finish(m.params.turn.status==='completed'?null:Error(m.params.turn.error?.message||'Composition interrupted'));
 }
 credentialStamp(){
  // Metadata only: never read, copy, log, or manage OAuth tokens ourselves.
  try{const s=statSync(path.join(process.env.CODEX_HOME||path.join(homedir(),'.codex'),'auth.json'));return `${s.ino}:${s.mtimeMs}:${s.size}`;}catch{return 'keychain-or-unavailable';}
 }
 async restart(){
  const old=this.proc;this.proc=null;this.ready=null;
  if(old&&old.exitCode===null)await new Promise(resolve=>{const timer=setTimeout(()=>{old.kill('SIGKILL');resolve();},2000);old.once('exit',()=>{clearTimeout(timer);resolve();});old.kill('SIGTERM');});
  for(const p of this.pending.values()){clearTimeout(p.timer);p.reject(Error('Codex connection restarted'));}this.pending.clear();
 }
 async compose(context,emit,signal){
  const stamp=this.credentialStamp();
  if(this.authStamp!==undefined&&stamp!==this.authStamp)await this.restart();
  this.authStamp=stamp;
  for(let attempt=0;attempt<2;attempt++){
   let delivered=0;
   try{
    await this.connect();
    const account=await this.rpc('account/read',{refreshToken:false});
    if(!account.account&&account.requiresOpenaiAuth)throw Error('Codex sign in required. Run codex login.');
    await this.composeOnce(context,event=>{if(event.type==='note')delivered++;emit(event);},signal);
    this.authStamp=this.credentialStamp();return;
   }catch(error){
    const authError=/access token|refresh token|sign in|logged out|unauthorized|401/i.test(error.message);
    if(!authError||attempt||delivered||signal.aborted)throw error;
    emit({type:'status',message:'Reconnecting Codex to reload the current login…'});
    await this.restart();this.authStamp=this.credentialStamp();
   }
  }
 }
 async composeOnce(context,emit,signal){
  await this.connect();if(signal.aborted)return;if(this.job)throw Error('Composer busy');
  const {audio,...analysis}=context;
  const instructions=`You are the musical director of a live band accompanying a guitarist. Compose an original four-bar phrase in 4/4 (16 beats). All musical content must be newly composed for this session, not a preset or quotation. Output only a JSON array of compact event rows:
[beat,instrumentCode,MIDI_or_chord_array,durationBeats,velocity,optionalRepeatCount,optionalRepeatEveryBeats].
The last two fields let YOU write recurring rhythmic motifs efficiently. Example syntax only: chord_array is an array of MIDI pitches; count includes the first event. Repetitions must start before beat16. Use 18-40 rows as needed, up to 256 expanded notes. Sort rows by their first beat. beat>=0 and <16, duration .1-4, velocity .15-.85. Final notes may sustain into the next phrase.
Palette: b=bass 28-52; k=electric keys, p=piano, g=plucked strings, s=strings, o=organ, v=vibraphone, f=flute, l=synth lead (48-84); d=drums 36 kick/38 snare/42 closed hat/46 open hat/49 crash. Choose the ensemble and density yourself; do not force every instrument into every phrase.
The attached audio is the actual unmodified recent playing, not a transcription. Listen to its rhythm, articulation, melodic contour and harmonic movement. Audio evidence takes precedence over uncertain estimates. Accompany that performance without replacing or rewriting the player's melody.
FOLLOW THE PLAYER FIRST: bpm is a provisional playback clock; attack estimates can confuse subdivisions with beats. Use the audio to choose subdivisions and accents, without blindly copying unreliable estimates. attackBeats gives measured strum timing in beats and attacksPerSecond describes activity. For rapid regular strumming, use a clear driving pulse and compatible subdivisions; avoid a slow ambient pad arrangement or a half-time backbeat. For sparse playing, leave room. Derive rhythmic accents from repeated observed attack spacing, while recognizing onset estimates are imperfect. Match the player's activity before adding elaborate melody. Keep the lead understated, prioritize supportive bass and rhythmic comping, and do not invent an unrelated tune. Use confidence on harmony frames: unstable evidence calls for less pitched density, not a confident busy solo.
COMPOSITION: Establish a memorable rhythmic identity, a singable short motif, purposeful bass movement and well-spaced harmony. Repetition with development is musical: preserve a strong groove, vary accents and phrase endings selectively. Do not scatter random drum hits or change instruments simply for novelty. Give each instrument a distinct register and role. Use voice leading, inversions, suspensions resolving to chord tones, syncopation and breathing room. Avoid all parts striking constantly together. Accompany rather than compete with the guitarist; melodic replies belong in spaces. Sustained voices can last several beats. Do not end every response like a separate song.
CONTINUITY: previous contains the full preceding composed phrase, including expanded rhythmic motifs. Do not copy the previous arrangement by default. Develop useful motifs with audible variation in phrasing, accents and voicing. A previous drums-only response is not a template to preserve, but current input rhythm and harmony always override stale previous notes. Do not preserve previous pitches against a changed guitar chord. phraseIndex indicates the next four-bar phrase in this session: shape a longer arc with restrained changes, contrasts and returns instead of resetting every call. The first phrase establishes the identity. Input attacks are approximate seconds relative to the latest five-second listening window, not an exact score. Use their density/spacing and energy to inform the groove. Harmony estimates may be uncertain; prefer stable recent evidence, avoid inventing a conflicting chord progression. Unknown estimated harmony does NOT require percussion-only output: infer harmony and melodic contour from the attached recording. Use sparse supportive pitched parts when the audio supports them. If the audio genuinely lacks a reliable pitched source, percussion alone is appropriate. All choices of notes, motifs, harmony voicings, rhythmic patterns and instrumentation are yours. No prose or tools. Emit immediately.`;
  const t=await this.rpc('thread/start',{model:'gpt-6.1-sol',cwd:this.cwd,ephemeral:true,sandbox:'read-only',approvalPolicy:'never',baseInstructions:instructions,config:{model_reasoning_effort:'low'}});
  if(signal.aborted){void this.rpc('thread/unsubscribe',{threadId:t.thread.id}).catch(()=>{});return;}
  await new Promise((resolve,reject)=>{
   let turnId,finished=false;
   const finish=error=>{if(finished)return;finished=true;clearTimeout(timeout);signal.removeEventListener('abort',abort);const count=this.job?.sent||0;this.job=null;void this.rpc('thread/unsubscribe',{threadId:t.thread.id}).catch(()=>{});if(error)reject(error);else{emit({type:'done',count,elapsedMs:Math.round(performance.now()-start)});resolve();}};
   const abort=()=>{if(turnId)void this.rpc('turn/interrupt',{threadId:t.thread.id,turnId}).catch(()=>{});finish(Error('Cancelled'));};
   const start=performance.now(),timeout=setTimeout(()=>{abort();},45000);
   this.job={threadId:t.thread.id,text:'',sent:0,emit,start,finish};signal.addEventListener('abort',abort,{once:true});
   this.rpc('turn/start',{threadId:t.thread.id,effort:'low',input:[{type:'text',text:JSON.stringify(analysis)},...(audio?[{type:'audio',url:audio}]:[])]}).then(r=>{turnId=r.turn.id;if(finished)void this.rpc('turn/interrupt',{threadId:t.thread.id,turnId}).catch(()=>{});}).catch(finish);
  });
 }
 close(){this.proc?.kill();}
}
